cmake_minimum_required(VERSION 3.20)
project(rc0 CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")
find_package(Threads REQUIRED)

# ---- rules: board, move generation, game end (no search, no evaluation) ----
add_library(rc0rules STATIC rules/geometry.cpp rules/position.cpp rules/movegen.cpp)
target_include_directories(rc0rules PUBLIC rules)
target_compile_options(rc0rules PRIVATE -Wall -Wextra)
target_link_libraries(rc0rules PUBLIC Threads::Threads)

# ---- engine core: input encoding + MCTS, independent of any NN library ----
add_library(rc0core STATIC src/encoding.cpp src/mcts.cpp src/uci4.cpp)
target_include_directories(rc0core PUBLIC src)
target_link_libraries(rc0core PUBLIC rc0rules)
target_compile_options(rc0core PRIVATE -Wall -Wextra)

# ---- LibTorch backend (from the PyTorch pip wheel; no CUDA toolkit needed) ----
option(RC0_TORCH "Build the LibTorch backend" ON)
set(RC0_BACKEND_SRC)
set(Torch_FOUND FALSE)
if(RC0_TORCH)
  if(NOT TORCH_DIR)
    find_program(RC0_PYTHON NAMES python3 python)
    execute_process(COMMAND ${RC0_PYTHON} -c "import torch,os;print(os.path.dirname(torch.__file__))"
                    OUTPUT_VARIABLE TORCH_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  if(EXISTS "${TORCH_DIR}/lib/libtorch.so")
    message(STATUS "LibTorch: ${TORCH_DIR}")
    set(Torch_FOUND TRUE)
    set(RC0_BACKEND_SRC src/backend_torch.cpp)
    add_library(torchwheel INTERFACE)
    target_include_directories(torchwheel INTERFACE ${TORCH_DIR}/include ${TORCH_DIR}/include/torch/csrc/api/include)
    set(_tl ${TORCH_DIR}/lib/libtorch.so ${TORCH_DIR}/lib/libtorch_cpu.so ${TORCH_DIR}/lib/libc10.so)
    if(EXISTS "${TORCH_DIR}/lib/libtorch_cuda.so")
      list(APPEND _tl ${TORCH_DIR}/lib/libtorch_cuda.so ${TORCH_DIR}/lib/libc10_cuda.so)
    endif()
    target_link_libraries(torchwheel INTERFACE ${_tl})
    target_compile_definitions(torchwheel INTERFACE _GLIBCXX_USE_CXX11_ABI=1)
  else()
    message(WARNING "LibTorch not found (pip install torch, or pass -DTORCH_DIR=...); building without a GPU backend")
  endif()
endif()

# ---- TensorRT backend (optional, ~2x LibTorch fp16). Needs NvInfer.h, libnvinfer and CUDA headers. ----
option(RC0_TRT "Build the TensorRT backend" ON)
set(Trt_FOUND FALSE)
if(RC0_TRT)
  find_path(TRT_INC NvInfer.h HINTS ${TRT_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/third_party/tensorrt/include /usr/include/x86_64-linux-gnu)
  if(NOT TRT_LIBDIR)
    execute_process(COMMAND ${RC0_PYTHON} -c "import tensorrt_libs,os;print(os.path.dirname(tensorrt_libs.__file__))"
                    OUTPUT_VARIABLE TRT_LIBDIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  find_library(TRT_LIB NAMES nvinfer libnvinfer.so.11 libnvinfer.so.10 HINTS ${TRT_LIBDIR})
  find_package(CUDAToolkit QUIET)
  if(TRT_INC AND TRT_LIB AND CUDAToolkit_FOUND)
    message(STATUS "TensorRT: ${TRT_INC} + ${TRT_LIB}")
    set(Trt_FOUND TRUE)
    list(APPEND RC0_BACKEND_SRC src/backend_trt.cpp)
    add_library(trtlib INTERFACE)
    target_include_directories(trtlib SYSTEM INTERFACE ${TRT_INC} ${CUDAToolkit_INCLUDE_DIRS})
    target_link_libraries(trtlib INTERFACE ${TRT_LIB} CUDA::cudart)
  else()
    message(STATUS "TensorRT not available; LibTorch only")
  endif()
endif()

add_executable(rc0 src/main.cpp src/selfplay.cpp ${RC0_BACKEND_SRC})
target_link_libraries(rc0 PRIVATE rc0core)
if(Torch_FOUND)
  target_link_libraries(rc0 PRIVATE torchwheel)
  target_compile_definitions(rc0 PRIVATE ZERO_HAVE_TORCH=1)
  target_link_options(rc0 PRIVATE -Wl,--no-as-needed)   # libtorch_cuda is needed though nothing references it
  set_target_properties(rc0 PROPERTIES BUILD_RPATH "${TORCH_DIR}/lib")
endif()
if(Trt_FOUND)
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(rc0 PRIVATE OpenMP::OpenMP_CXX)
  endif()
  target_link_libraries(rc0 PRIVATE trtlib)
  target_compile_definitions(rc0 PRIVATE ZERO_HAVE_TRT=1)
  get_filename_component(_trtdir ${TRT_LIB} DIRECTORY)
  set_property(TARGET rc0 APPEND PROPERTY BUILD_RPATH "${_trtdir}")
endif()

# ---- tests ----
enable_testing()
add_executable(test_rules tests/test_rules.cpp)
target_link_libraries(test_rules PRIVATE rc0rules)
add_executable(test_encoding tests/test_encoding.cpp)
target_link_libraries(test_encoding PRIVATE rc0core)
add_executable(test_mcts tests/test_mcts.cpp)
target_link_libraries(test_mcts PRIVATE rc0core)
add_test(NAME rules COMMAND test_rules)
add_test(NAME encoding COMMAND test_encoding)
add_test(NAME mcts COMMAND test_mcts)
